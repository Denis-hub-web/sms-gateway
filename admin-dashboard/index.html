Developer Integration Guide

Link your website, application, or CRM to this SMS Gateway in 3 lines of code

1. Authentication & Endpoint

Send an HTTP POST request to your gateway server endpoint. Include your API Key in the X-API-Key request header:

Endpoint Specification POST 
http://192.168.100.5:8080/api/v1/sms/send

2. Code Examples

curl -X POST http://192.168.100.5:8080/api/v1/sms/send \
  -H "X-API-Key: YOUR_API_KEY_HERE" \
  -H "Content-Type: application/json" \
  -d '{
        "phoneNumber": "+25575855909",
        "message": "Hello from external system!"
      }'
const fetch = require('node-fetch');

async function sendSms() {
  const response = await fetch('http://192.168.100.5:8080/api/v1/sms/send', {
    method: 'POST',
    headers: {
      'X-API-Key': 'YOUR_API_KEY_HERE',
      'Content-Type': 'application/json'
    },
    body: JSON.stringify({
      phoneNumber: '+25575855909',
      message: 'Hello from Node.js!'
    })
  });
  const data = await response.json();
  console.log(data);
}

sendSms();
import requests

url = "http://192.168.100.5:8080/api/v1/sms/send"
headers = {
    "X-API-Key": "YOUR_API_KEY_HERE",
    "Content-Type": "application/json"
}
payload = {
    "phoneNumber": "+25575855909",
    "message": "Hello from Python!"
}

response = requests.post(url, json=payload, headers=headers)
print(response.json())
<?php
$ch = curl_init('http://192.168.100.5:8080/api/v1/sms/send');
$payload = json_encode([
    "phoneNumber" => "+25575855909",
    "message" => "Hello from PHP!"
]);
curl_setopt($ch, CURLOPT_POSTFIELDS, $payload);
curl_setopt($ch, CURLOPT_HTTPHEADER, [
    'X-API-Key: YOUR_API_KEY_HERE',
    'Content-Type: application/json'
]);
curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
$result = curl_exec($ch);
curl_close($ch);
echo $result;
?>
